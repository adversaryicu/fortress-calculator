Step 1: Extract Skills Data Run in Game Console

Open your game tab, press F12 to open DevTools, select the Console tab, paste this script, and hit Enter. It audits your exact success rates, unlocks, and bag capacity, then automatically copies the JSON to your clipboard. 

(() => {
    if (typeof ClientState === 'undefined' || !ClientState.skills) {
        console.error("ClientState or skills not loaded yet. Make sure you are logged into the game.");
        return;
    }

    // --- 1. FORTRESS SKILLING AUDIT ---
    const roles = [
        { id: 'fish',   gather: 'fish',   process: 'cook',  stat: 'Health',       gatherNodeId: 'gauntlet_fish',   processNodeId: 'gauntlet_cook' },
        { id: 'mine',   gather: 'mine',   process: 'smith', stat: 'Melee Force',  gatherNodeId: 'gauntlet_mine',   processNodeId: 'gauntlet_smith' },
        { id: 'forage', gather: 'forage', process: 'craft', stat: 'Range Force',  gatherNodeId: 'gauntlet_forage', processNodeId: 'gauntlet_craft' },
        { id: 'invoke', gather: 'invoke', process: 'imbue', stat: 'Magic Force',  gatherNodeId: 'gauntlet_invoke', processNodeId: 'gauntlet_imbue' },
        { id: 'farm',   gather: 'farm',   process: 'brew',  stat: 'Defense',      gatherNodeId: 'gauntlet_farm',   processNodeId: 'gauntlet_brew' }
    ];

    const getLevel = (s) => (ClientState.skills[s] && ClientState.skills[s].level) || 1;
    const getTier = (pairLvl) => pairLvl >= 100 ? 10 : Math.floor(pairLvl / 10);
    const getReqLevel = (tier) => (tier === 0 ? 1 : tier * 10);
    const haste = (typeof Constants !== 'undefined' && Constants.GAUNTLET_HASTE) ? Constants.GAUNTLET_HASTE : 10;

    const getSuccess = (skill, req) => {
        if (typeof ClientState.successChanceFor === 'function') return ClientState.successChanceFor(skill, req);
        const lvl = getLevel(skill);
        return (typeof Constants !== 'undefined' && Constants.successRate) 
            ? Constants.successRate(lvl, req, 0) / 100 
            : Math.min(1, (70 + Math.max(0, lvl - req)) / 100);
    };

    const getDoubleOrSave = (skill, req) => {
        if (typeof ClientState.doubleGatherOrSaveFor === 'function') return ClientState.doubleGatherOrSaveFor(skill, req);
        return 0;
    };

    const getDuration = (skill, req) => {
        if (typeof ClientState.actionDurationFor === 'function') return ClientState.actionDurationFor(skill, req);
        const lvl = getLevel(skill);
        const base = (typeof Constants !== 'undefined' && Constants.actionDuration) 
            ? Constants.actionDuration(lvl, req) 
            : (10 - (lvl - 1) * 0.01);
        const vigorLvl = (ClientState.skills && ClientState.skills.vigor && ClientState.skills.vigor.level) || 1;
        const vigorRed = (typeof Constants !== 'undefined' && Constants.vigorActionReduction)
            ? Constants.vigorActionReduction(vigorLvl)
            : 0;
        return Math.max(2.0, base - vigorRed);
    };

    const bagSize = (typeof ClientState.getBagCapacity === 'function') ? ClientState.getBagCapacity() : 28;
    const baseWalkSpeed = (typeof Constants !== 'undefined' && Constants.BASE_MOVEMENT_SPEED) ? Constants.BASE_MOVEMENT_SPEED : 3.0;
    const trainersBonus = (typeof ClientState.getTrainersMoveSpeedBonus === 'function') ? ClientState.getTrainersMoveSpeedBonus() : 0;
    const beaconBonus = (typeof ClientState.getBeaconMoveSpeedBonus === 'function') ? ClientState.getBeaconMoveSpeedBonus() : 0;
    const ancientBonus = (typeof ClientState.getShardSetMoveSpeedBonus === 'function') ? ClientState.getShardSetMoveSpeedBonus() : 0;
    const liveHostSpeed = baseWalkSpeed + trainersBonus + beaconBonus + ancientBonus;
    const liveCloneSpeed = liveHostSpeed * 0.9;

    const roleData = [];
    roles.forEach(r => {
        const pairLevel = getLevel(r.gather) + getLevel(r.process);
        const tier = getTier(pairLevel);
        const reqLvl = getReqLevel(tier);
        const tierVal = (typeof Constants !== 'undefined' && Constants.GAUNTLET_TIER_STAT_VALUES) 
            ? (Constants.GAUNTLET_TIER_STAT_VALUES[tier] || (tier + 2)) 
            : (tier === 0 ? 2 : tier + 2);

        const gRate = getSuccess(r.gather, reqLvl);
        const pRate = getSuccess(r.process, reqLvl);
        const gDouble = getDoubleOrSave(r.gather, reqLvl);
        const pSave = getDoubleOrSave(r.process, reqLvl);
        const gActionTime = Number((getDuration(r.gather, reqLvl) / haste).toFixed(3));
        const pActionTime = Number((getDuration(r.process, reqLvl) / haste).toFixed(3));

        const gatherItemsPerAction = 1 + gDouble;
        const gatherAttempts = bagSize / (gRate * gatherItemsPerAction);
        const gatherTime = gatherAttempts * gActionTime;
        const processItemsConsPerAction = 1 - pSave;
        const processAttempts = bagSize / processItemsConsPerAction;
        const processTime = processAttempts * pActionTime;
        const expectedItems = bagSize * pRate;
        const expectedStat = expectedItems * tierVal;

        const walkDist = (typeof Constants !== 'undefined' && Constants.GAUNTLET_NODES && Constants.GAUNTLET_NODES[r.gatherNodeId]) 
            ? Constants.GAUNTLET_NODES[r.gatherNodeId].walkDist 
            : 300;
        const transitHost = (walkDist * 2) / liveHostSpeed;
        const cycleTimeHost = Number((gatherTime + processTime + transitHost).toFixed(1));

        roleData.push({
            role: r.stat,
            gatherSkill: r.gather,
            gatherLevel: getLevel(r.gather),
            gatherSuccessPct: Number((gRate * 100).toFixed(1)),
            gatherDoublePct: Number((gDouble * 100).toFixed(1)),
            gatherActionTime: gActionTime,
            processSkill: r.process,
            processLevel: getLevel(r.process),
            processSuccessPct: Number((pRate * 100).toFixed(1)),
            processSavePct: Number((pSave * 100).toFixed(1)),
            processActionTime: pActionTime,
            activeTier: tier,
            tierRequiredLevel: reqLvl,
            statValuePerItem: tierVal,
            expectedItemsFinished: expectedItems,
            bagCapacity: bagSize,
            expectedStatYield: expectedStat,
            maxStatYield: bagSize * tierVal,
            walkDistance: walkDist,
            cycleTimeHost: cycleTimeHost,
            efficiencyPct: Number(((expectedStat / (bagSize * tierVal)) * 100).toFixed(1))
        });
    });

    // --- 2. COMBAT AUDIT FOR DUNGEONS ---
    const combatSkillsList = ['strike', 'heft', 'sight', 'draw', 'focus', 'surge', 'dodge', 'defend', 'vigor', 'life', 'spirit', 'slay', 'delve'];
    const combatSkills = {};
    combatSkillsList.forEach(s => { combatSkills[s] = getLevel(s); });

    const combatLoadouts = {};
    if (ClientState.equipmentLoadouts && typeof Constants !== 'undefined') {
        const loadoutKeys = Constants.COMBAT_LOADOUTS || Object.keys(ClientState.equipmentLoadouts);
        loadoutKeys.forEach(k => {
            const eq = ClientState.equipmentLoadouts[k];
            if (eq && Object.values(eq).some(Boolean)) {
                const stats = typeof ClientState.computeLoadoutStats === 'function' ? ClientState.computeLoadoutStats(k) : null;
                const name = typeof ClientState.loadoutName === 'function' ? ClientState.loadoutName(k) : k;
                const resolvedStyle = typeof ClientState.getResolvedCombatStyle === 'function' ? ClientState.getResolvedCombatStyle(k) : (stats ? stats.attackStyle : 'melee');
                combatLoadouts[k] = {
                    id: k,
                    name: name,
                    style: resolvedStyle,
                    stats: stats || {}
                };
            }
        });
    }

    const foods = [];
    if (ClientState.bank && ClientState.itemsData) {
        for (const [itemId, qty] of Object.entries(ClientState.bank)) {
            if (qty > 0 && ClientState.itemsData[itemId] && ClientState.itemsData[itemId].healHP) {
                const it = ClientState.itemsData[itemId];
                foods.push({
                    id: itemId,
                    name: it.name,
                    tier: it.tier,
                    healHP: it.healHP,
                    qty: qty
                });
            }
        }
        foods.sort((a, b) => b.healHP - a.healHP);
    }

    const activePotions = typeof ClientState.getPotionBoosts === 'function' ? ClientState.getPotionBoosts() : {};
    let respawnReduction = 0;
    if (typeof ClientState.getSlateStats === 'function') {
        respawnReduction += (ClientState.getSlateStats().respawnReduction || 0);
    }

    const report = {
        player: ClientState.playerName || 'Player',
        partySize: 5,
        bagCapacity: bagSize,
        hostSpeed: liveHostSpeed,
        cloneSpeed: liveCloneSpeed,
        haste: haste,
        roles: roleData,
        combatSkills: combatSkills,
        combatLoadouts: combatLoadouts,
        foods: foods,
        activePotions: activePotions,
        respawnReduction: respawnReduction
    };

    const jsonText = JSON.stringify(report, null, 2);
    window.__fortressData = report;
    window.__fortressJson = jsonText;

    let copied = false;
    try {
        if (typeof copy === 'function') {
            copy(jsonText);
            copied = true;
        }
    } catch (e) {}

    if (!copied && navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(jsonText).then(() => { copied = true; }).catch(() => {});
    }

    console.table(report.roles);
    if (copied) {
        console.log("%c>>> Auto-copied to clipboard! Paste into the Simulator! <<<", "color: #5cb85c; font-weight: bold; font-size: 13px;");
    } else {
        console.log("%c>>> To copy to clipboard, type: copy(__fortressJson) and press Enter <<<", "color: #38bdf8; font-weight: bold; font-size: 13px;");
    }
    return report;
})();
