Step 1: Extract Skills Data Run in Game Console

Open your game tab, press F12 to open DevTools, select the Console tab, paste this script, and hit Enter. It audits your exact success rates, unlocks, and bag capacity, then automatically copies the JSON to your clipboard. 

(() => {
    if (typeof ClientState === 'undefined' || !ClientState.skills) {
        console.error("ClientState or skills not loaded yet. Make sure you are logged into the game.");
        return;
    }

    const roles = [
        { id: 'fish',   gather: 'fish',   process: 'cook',  stat: 'Health',       gatherNodeId: 'gauntlet_fish',   processNodeId: 'gauntlet_cook' },
        { id: 'mine',   gather: 'mine',   process: 'smith', stat: 'Melee Force',  gatherNodeId: 'gauntlet_mine',   processNodeId: 'gauntlet_smith' },
        { id: 'forage', gather: 'forage', process: 'craft', stat: 'Range Force',  gatherNodeId: 'gauntlet_forage', processNodeId: 'gauntlet_craft' },
        { id: 'invoke', gather: 'invoke', process: 'imbue', stat: 'Magic Force',  gatherNodeId: 'gauntlet_invoke', processNodeId: 'gauntlet_imbue' },
        { id: 'farm',   gather: 'farm',   process: 'brew',  stat: 'Defense',      gatherNodeId: 'gauntlet_farm',   processNodeId: 'gauntlet_brew' }
    ];

    const getLevel = (s) => (ClientState.skills[s] && ClientState.skills[s].level) || 1;
    const getTier = (pairLvl) => pairLvl >= 100 ? 10 : Math.floor(pairLvl / 10);
    const getReqLevel = (tier) => (tier === 0 ? 1 : tier * 10);
    const haste = (typeof Constants !== 'undefined' && Constants.GAUNTLET_HASTE) ? Constants.GAUNTLET_HASTE : 10;

    const getSuccess = (skill, req) => {
        if (typeof ClientState.successChanceFor === 'function') return ClientState.successChanceFor(skill, req);
        const lvl = getLevel(skill);
        return (typeof Constants !== 'undefined' && Constants.successRate) 
            ? Constants.successRate(lvl, req, 0) / 100 
            : Math.min(1, (70 + Math.max(0, lvl - req)) / 100);
    };

    const getDoubleOrSave = (skill, req) => {
        if (typeof ClientState.doubleGatherOrSaveFor === 'function') return ClientState.doubleGatherOrSaveFor(skill, req);
        return 0;
    };

    const getDuration = (skill, req) => {
        if (typeof ClientState.actionDurationFor === 'function') return ClientState.actionDurationFor(skill, req);
        const lvl = getLevel(skill);
        const base = (typeof Constants !== 'undefined' && Constants.actionDuration) 
            ? Constants.actionDuration(lvl, req) 
            : (10 - (lvl - 1) * 0.01);
        const vigorLvl = (ClientState.skills && ClientState.skills.vigor && ClientState.skills.vigor.level) || 1;
        const vigorRed = (typeof Constants !== 'undefined' && Constants.vigorActionReduction)
            ? Constants.vigorActionReduction(vigorLvl)
            : (vigorLvl + 1) * 0.005;
        const minDur = (typeof Constants !== 'undefined' && Constants.MIN_ACTION_DURATION) || 2.0;
        return Math.max(minDur, base - vigorRed);
    };

    const getDepositVal = (tier) => {
        if (typeof Constants !== 'undefined' && Constants.gauntletDepositValue) return Constants.gauntletDepositValue(tier);
        return Math.round(2 * Math.pow(1.35, tier));
    };

    const satchelCount = (ClientState.unlocks && ClientState.unlocks.satchels) 
        ? Object.keys(ClientState.unlocks.satchels).filter(k => ClientState.unlocks.satchels[k]).length : 0;
    const bagSize = (typeof Constants !== 'undefined' && Constants.INVENTORY_SLOTS ? Constants.INVENTORY_SLOTS : 28) 
        + satchelCount * (typeof Constants !== 'undefined' && Constants.SATCHEL_SLOTS_PER ? Constants.SATCHEL_SLOTS_PER : 4);

    const calcArenaWalkDistance = (roleDef) => {
        try {
            if (typeof ClientState.getPathDistance === 'function') {
                const d1 = ClientState.getPathDistance('gauntlet_raid', roleDef.gatherNodeId);
                const d2 = ClientState.getPathDistance(roleDef.gatherNodeId, roleDef.processNodeId);
                const d3 = ClientState.getPathDistance(roleDef.processNodeId, 'gauntlet_raid');
                if (d1 > 0 && d2 > 0 && d3 > 0) return Math.round(d1 + d2 + d3);
            }
            const nodes = ClientState.nodesData || [];
            const byId = {};
            for (let i = 0; i < nodes.length; i++) byId[nodes[i].id] = nodes[i];
            const center = byId['gauntlet_raid'];
            const gNode = byId[roleDef.gatherNodeId];
            const pNode = byId[roleDef.processNodeId];
            if (center && gNode && pNode) {
                const d = (a, b) => Math.hypot(a.position.x - b.position.x, a.position.y - b.position.y);
                return Math.round(d(center, gNode) + d(gNode, pNode) + d(pNode, center));
            }
        } catch(e) {}
        return 610;
    };

    const calcLiveMovementSpeed = () => {
        const vigorLevel = (ClientState.skills && ClientState.skills.vigor && ClientState.skills.vigor.level) || 1;
        let vigorEffLevel = vigorLevel;
        if (typeof EffectiveLevel !== 'undefined' && typeof EffectiveLevel.compute === 'function') {
            try {
                const vigorEff = EffectiveLevel.compute({
                    level: vigorLevel,
                    pets: ClientState.pets,
                    capes: ClientState.capes,
                    tools: ClientState.unlocks && ClientState.unlocks.tools
                });
                if (vigorEff && vigorEff.effective) vigorEffLevel = vigorEff.effective;
            } catch(e) {}
        }

        const unl = ClientState.unlocks || {};
        const relics = unl.relics || {};
        let telBonus = 0;
        const telPer = (typeof Constants !== 'undefined' && Constants.TELESCOPE_MOVE_SPEED_BONUS_PER) || 0.005;
        const telShinyPer = (typeof Constants !== 'undefined' && Constants.TELESCOPE_SHINY_MOVE_SPEED_BONUS_PER) || 0.01;
        for (let t = 0; t <= 10; t++) {
            if (relics['telescope_t' + t]) telBonus += telPer;
            if (relics['shiny_telescope_t' + t]) telBonus += telShinyPer;
        }

        const beacons = unl.beacons || {};
        let beaconBonus = 0;
        const bTiers = (typeof Constants !== 'undefined' && Constants.GAUNTLET_RARE_TIER_KEYS) || ['feeble','dire','savage','fell','ancient','chaos','cataclysm'];
        for (let b = 0; b < bTiers.length; b++) {
            if (beacons['beacon_swift_' + bTiers[b]]) {
                beaconBonus += (typeof Constants !== 'undefined' && typeof Constants.swiftBeaconMoveBonus === 'function')
                    ? Constants.swiftBeaconMoveBonus(b) : 0.01 * (b + 1);
            }
        }

        const compasses = unl.compasses || {};
        let compassCount = 0;
        for (let c = 0; c <= 10; c++) {
            if (compasses['enchanted_compass_t' + c]) compassCount++;
        }
        const compassBonus = compassCount * ((typeof Constants !== 'undefined' && Constants.COMPASS_MOVE_SPEED_BONUS_PER) || 0.005);

        let gauntBonus = 0;
        const banners = unl.banners || {};
        const ledgers = unl.ledgers || {};
        for (let g = 0; g < bTiers.length; g++) {
            if (typeof Constants !== 'undefined' && typeof Constants.gauntletMilestoneMoveBonus === 'function') {
                if (banners['gauntlet_banner_' + bTiers[g]]) gauntBonus += Constants.gauntletMilestoneMoveBonus(g);
                if (ledgers['gauntlet_ledger_' + bTiers[g]]) gauntBonus += Constants.gauntletMilestoneMoveBonus(g);
            }
        }

        let talentBonus = 0;
        if (window.RaidTreeEffects && ClientState.raidTrees) {
            try {
                const mte = window.RaidTreeEffects.sumEffects({
                    delve: [],
                    fortify: (ClientState.raidTrees.fortify && ClientState.raidTrees.fortify.allocated) || []
                });
                talentBonus = (mte && mte.moveSpeedPct ? mte.moveSpeedPct / 100 : 0);
            } catch(e) {}
        }

        const collectorBonus = ((typeof ClientState.getCollectorProgress === 'function' ? ClientState.getCollectorProgress() : 0)
            * ((typeof Constants !== 'undefined' && Constants.COLLECTOR_MAX_MOVE_SPEED) || 0.05));
        let potionMult = 0;
        if (ClientState.activePotions && ClientState.activePotions.movement && ClientState.activePotions.movement.length > 0) {
            const potId = ClientState.activePotions.movement[0].potionId;
            if (typeof Constants !== 'undefined' && Constants.MOVEMENT_POTION_LOOKUP && Constants.MOVEMENT_POTION_LOOKUP[potId]) {
                potionMult = Constants.MOVEMENT_POTION_LOOKUP[potId].speedMultiplier || 0;
            }
        }

        // If player is currently in the Fortress raid lobby, read exact server steps/sec directly
        if (ClientState.raidLobby && ClientState.raidLobby.members) {
            const me = ClientState.raidLobby.members.find(m => m.playerId === ClientState.playerId);
            if (me && me.move) {
                if (me.move.stepsWithPotion != null) return Number(me.move.stepsWithPotion.toFixed(1));
                if (me.move.steps != null) return Number(me.move.steps.toFixed(1));
            }
        }

        const ancientBonus = (typeof ClientState.getShardSetMoveSpeedBonus === 'function')
            ? ClientState.getShardSetMoveSpeedBonus()
            : 0;

        const mvOpts = {
            vigorLevel: vigorEffLevel,
            telescope: telBonus,
            compass: compassBonus,
            gauntlet: gauntBonus,
            beacon: beaconBonus,
            collector: collectorBonus,
            ancient: ancientBonus,
            talent: talentBonus,
            potionMult: potionMult
        };

        if (typeof Constants !== 'undefined' && typeof Constants.totalMovementSpeed === 'function') {
            return Number(Constants.totalMovementSpeed(mvOpts).toFixed(1));
        }
        if (typeof Constants !== 'undefined' && typeof Constants.movementSpeed === 'function') {
            return Number(Constants.movementSpeed(vigorEffLevel).toFixed(1));
        }
        return Number((10 + (vigorEffLevel + 1) * (0.1 + 0.0005 * vigorEffLevel)).toFixed(1));
    };

    const hostSpeed = calcLiveMovementSpeed();
    const cloneMult = (typeof Constants !== 'undefined' && Constants.GAUNTLET_CLONE_MOVE_MULT) ? Constants.GAUNTLET_CLONE_MOVE_MULT : 0.9;
    const cloneSpeed = Number((hostSpeed * cloneMult).toFixed(1));

    const report = {
        player: ClientState.playerName || 'Player',
        partySize: 5,
        bagCapacity: bagSize,
        hostSpeed: hostSpeed,
        cloneSpeed: cloneSpeed,
        haste: haste,
        walkDistance: 610,
        skills: ClientState.skills,
        roles: []
    };

    roles.forEach(r => {
        const gLvl = getLevel(r.gather);
        const pLvl = getLevel(r.process);
        const tier = getTier(Math.min(gLvl, pLvl));
        const reqLvl = getReqLevel(tier);
        const gSuccess = getSuccess(r.gather, reqLvl);
        const pSuccess = getSuccess(r.process, reqLvl);
        const gDouble = getDoubleOrSave(r.gather, 1);
        const pSave = getDoubleOrSave(r.process, 1);
        const gRawDuration = getDuration(r.gather, 1);
        const pRawDuration = getDuration(r.process, 1);
        const gActionTime = Number((gRawDuration / haste).toFixed(3));
        const pActionTime = Number((pRawDuration / haste).toFixed(3));
        const tierVal = getDepositVal(tier);
        const walkDist = calcArenaWalkDistance(r);

        const effectiveConversion = Math.min(1, pSuccess / (1 - pSave * pSuccess));
        const expectedItems = Number((bagSize * effectiveConversion).toFixed(1));
        const expectedStat = Number((expectedItems * tierVal).toFixed(1));

        // Cycle time breakdown
        const attemptsGather = bagSize / (gSuccess * (1 + gDouble));
        const gatherTime = attemptsGather * gActionTime;
        const attemptsProcess = bagSize / Math.max(0.01, 1 - pSave * pSuccess);
        const processTime = attemptsProcess * pActionTime;
        const walkTimeHost = walkDist / hostSpeed;
        const cycleTimeHost = Number((walkTimeHost + gatherTime + processTime).toFixed(1));

        report.roles.push({
            role: r.stat,
            gatherSkill: r.gather,
            gatherLevel: gLvl,
            gatherSuccessPct: Number((gSuccess * 100).toFixed(1)),
            gatherDoublePct: Number((gDouble * 100).toFixed(1)),
            gatherActionTime: gActionTime,
            processSkill: r.process,
            processLevel: pLvl,
            processSuccessPct: Number((pSuccess * 100).toFixed(1)),
            processSavePct: Number((pSave * 100).toFixed(1)),
            processActionTime: pActionTime,
            activeTier: tier,
            tierRequiredLevel: reqLvl,
            statValuePerItem: tierVal,
            expectedItemsFinished: expectedItems,
            bagCapacity: bagSize,
            expectedStatYield: expectedStat,
            maxStatYield: bagSize * tierVal,
            walkDistance: walkDist,
            cycleTimeHost: cycleTimeHost,
            efficiencyPct: Number(((expectedStat / (bagSize * tierVal)) * 100).toFixed(1))
        });
    });

    const jsonText = JSON.stringify(report, null, 2);
    window.__fortressData = report;
    window.__fortressJson = jsonText;

    let copied = false;

    // 1. Remove any lingering floating pill element if present
    try {
        const oldPill = document.getElementById('fortress-copy-pill');
        if (oldPill) oldPill.remove();
    } catch (e) {}

    // 2. Try DevTools native copy() command
    try {
        if (typeof copy === 'function') {
            copy(jsonText);
            copied = true;
        }
    } catch (e) {}

    // 3. Try classic off-screen execCommand
    if (!copied) {
        try {
            const ta = document.createElement('textarea');
            ta.value = jsonText;
            ta.style.position = 'fixed';
            ta.style.left = '-9999px';
            document.body.appendChild(ta);
            ta.focus();
            ta.select();
            copied = document.execCommand('copy');
            document.body.removeChild(ta);
        } catch (e) {}
    }

    // 4. Try modern navigator.clipboard with safe .catch()
    if (!copied && navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(jsonText).then(() => {
            copied = true;
        }).catch(() => {});
    }

    console.table(report.roles);
    if (copied) {
        console.log("%c>>> Auto-copied to clipboard! Paste into the Fortress Simulator! <<<", "color: #5cb85c; font-weight: bold; font-size: 13px;");
    } else {
        console.log("%c>>> To copy to clipboard, type: copy(__fortressJson) and press Enter <<<", "color: #38bdf8; font-weight: bold; font-size: 13px;");
    }
    return report;
})();
